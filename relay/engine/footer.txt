  const drain = async () => { let count = 0; while (deferred.length && count++ < 12) await deferred.shift()(); };

  };
  const call = callJson;
  callJson = async (kind, ...args) => {
    try {
      const result = await call(kind, ...args);
      if (["check", "read"].includes(kind) && !Array.isArray(result?.issues)) throw { code: "invalid_check", message: "검사 응답 형식이 올바르지 않습니다." };
      if (kind === "fix" && !Array.isArray(result?.edits)) throw { code: "invalid_fix", message: "수정 응답 형식이 올바르지 않습니다." };
      if (kind === "analyze" && !Array.isArray(result?.facts)) throw { code: "invalid_facts", message: "확정 사실 응답 형식이 올바르지 않습니다." };
      return result;
    } catch (e) { if (e?.code !== "pipeline_yield") runtime.failures.push({ kind, code: e?.code || "engine_error" }); throw e; }
  };

      return { world: copy(w), pages: copy(changed), removed, page: page ? copy(page) : null, targetKey,
        checked: !!page?.check && !required.some(f => ["check", "read"].includes(f.kind)), fixed: page?.autoFixed?.n || 0, remaining: page?.check?.issues?.length || 0,
        notFound: S.nf.has(k), redirectTo: redirect || null, sceneUsed: outcome?.sceneUsed || "", failures: required, error,
        warnings: runtime.failures.filter(f => !required.includes(f)) };
    }
  };
